```js script.js
const dateOf = (ctx) => /^\d{4}-\d{2}-\d{2}$/.test(ctx.name) ? ctx.name : kb.today();
kb.macro("date", (arg, ctx) => dateOf(ctx));
kb.macro("hello", (arg) => `こんにちは、${arg || "名無し"}`);
// {{embed ページ#見出し}}。本文は kb.page で読むので async
kb.macro("embed", async (arg, ctx) => {
  const i = arg.lastIndexOf("#");
  const [ref, name] = i < 0 ? [arg, null] : [arg.slice(0, i), arg.slice(i + 1).trim()];
  const found = kb.resolve(ref);
  if (!found) return `（embed: ページ「${ref.trim()}」がない）`;
  if (ctx.stack.includes(found[0])) return `（embed: 「${found[0]}」が自分自身を取り込んでいる）`;
  const page = await kb.page(found[0]);
  if (!page) return `（embed: 「${found[0]}」を読めない）`;
  const md = name ? kb.section(page.body, name) : page.body.replace(/^# .*\n/m, "").trim();
  if (md === null) return `（embed: 「${found[0]}」に見出し「${name}」がない）`;
  return kb.expand(md, { ...ctx, stack: [...ctx.stack, found[0]] });
});
// {{touched}}: その日に作成・更新したページ
kb.macro("touched", (arg, ctx) => {
  const date = dateOf(ctx);
  // kb.pages の順は控えの順なので、名前で並べてから出す
  const hits = [...kb.pages].filter(([name, p]) => name !== date && (p.created === date || p.updated === date)).sort(([a], [b]) => (a < b ? -1 : a > b ? 1 : 0));
  return hits.length ? hits.map(([name, p]) => `- ${kb.wikilink(name, p)}`).join("\n") : "（なし）";
});
```
